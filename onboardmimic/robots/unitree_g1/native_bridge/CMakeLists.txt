cmake_minimum_required(VERSION 3.10)
project(onboardmimic_bridge)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
add_executable(core_test core_test.cpp)
add_executable(authority_test authority_test.cpp)
target_compile_options(authority_test PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME watchdog_core COMMAND core_test)
add_test(NAME exclusive_writer_authority COMMAND authority_test)
option(BUILD_ROS_BRIDGE "Build ROS2 readers and command bridge" ON)
if(BUILD_ROS_BRIDGE)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(unitree_hg REQUIRED)
  find_package(Threads REQUIRED)
  foreach(target sonic_state_reader sonic_command_bridge)
    add_executable(${target} ${target}.cpp)
    ament_target_dependencies(${target} rclcpp unitree_hg)
    target_link_libraries(${target} zmq Threads::Threads)
  endforeach()
endif()
